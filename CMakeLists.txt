cmake_minimum_required(VERSION 3.20)

project(prosperoweb LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

if(CMAKE_CROSSCOMPILING)
  set(_prosperoweb_tests_default OFF)
else()
  set(_prosperoweb_tests_default ON)
endif()
option(PROSPEROWEB_BUILD_TESTS "Build unit tests" ${_prosperoweb_tests_default})

set(PROSPEROWEB_FRONTEND "none" CACHE STRING "Frontend to build: none, host-gtk or ps5")
set_property(CACHE PROSPEROWEB_FRONTEND PROPERTY STRINGS none host-gtk ps5)
if(NOT PROSPEROWEB_FRONTEND MATCHES "^(none|host-gtk|ps5)$")
  message(FATAL_ERROR "PROSPEROWEB_FRONTEND must be one of: none, host-gtk, ps5 (got '${PROSPEROWEB_FRONTEND}')")
endif()

set(PROSPEROWEB_THIRD_PARTY_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party")

add_library(prosperoweb_warnings INTERFACE)
target_compile_options(prosperoweb_warnings INTERFACE -Wall -Wextra -Werror)

add_subdirectory(src/core)
add_subdirectory(src/shell)
add_subdirectory(src/ps5/spawn)
add_subdirectory(src/ps5/audio)
add_subdirectory(src/ps5/display)
add_subdirectory(src/ps5/video)
add_subdirectory(src/ps5/cpu)
add_subdirectory(src/ps5/evohost)
add_subdirectory(src/chrome-ui)

# Host-only build tool: the ad-block list -> rule-list converter.
if(NOT CMAKE_CROSSCOMPILING AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tools/addons_convert/CMakeLists.txt")
  add_subdirectory(tools/addons_convert)
endif()

if(PROSPEROWEB_FRONTEND STREQUAL "host-gtk")
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/frontend/host-gtk/CMakeLists.txt")
    add_subdirectory(src/frontend/host-gtk)
  else()
    message(FATAL_ERROR "host-gtk frontend is not available yet")
  endif()
elseif(PROSPEROWEB_FRONTEND STREQUAL "ps5")
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/frontend/ps5/CMakeLists.txt")
    add_subdirectory(src/frontend/ps5)
  else()
    message(FATAL_ERROR "ps5 frontend is not available yet")
  endif()
endif()

if(PROSPEROWEB_BUILD_TESTS)
  enable_testing()
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/core/CMakeLists.txt")
    add_subdirectory(tests/core)
  endif()
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/shell/CMakeLists.txt")
    add_subdirectory(tests/shell)
  endif()
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/ps5/CMakeLists.txt")
    add_subdirectory(tests/ps5)
  endif()
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/chrome-ui/CMakeLists.txt")
    add_subdirectory(tests/chrome-ui)
  endif()
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/media/CMakeLists.txt")
    add_subdirectory(tests/media)
  endif()
endif()
